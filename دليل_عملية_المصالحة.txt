# دليل شامل لعملية طلب المصالحة
# Reconcile Request Process Guide

================================================================================
📋 نظرة عامة على العملية
================================================================================
يوضح هذا الدليل خطوات تنفيذ طلب المصالحة من البداية حتى النهاية مع تحديد
الأدوار والمسؤوليات والـ Endpoints المستخدمة.

تسلسل الحالات:
طلب جديد → إحالة للمشرف → إحالة للمستشار → قيد التنفيذ →
مراجعة المشرف → إكمال نهائي

================================================================================
👥 الأدوار والصلاحيات
================================================================================

الدور: Anonymous
الصلاحيات: إنشاء طلب المصالحة فقط

الدور: Admin
الصلاحيات: عرض جميع الطلبات، إحالة للمشرفين، إلغاء، إكمال نهائي

الدور: Supervisor
الصلاحيات: عرض طلباته، إحالة للمستشارين، مراجعة الطلبات

الدور: Mediation
الصلاحيات: عرض طلباته، بدء التنفيذ، إكمال المصالحة، إرسال SMS

================================================================================
🔄 المرحلة 1: إرسال الطلب (العميل)
================================================================================

الخطوة 1.1: إنشاء طلب المصالحة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Anonymous (لا يحتاج تسجيل دخول)
الحالة: NewRequest
الـ Endpoint: POST api/reconcilerequest

الوصف: يقوم العميل بإرسال طلب المصالحة مع البيانات الأساسية

البيانات المطلوبة:
{
    "name": "اسم الطالب",
    "email": "email@example.com",
    "phoneNumber": "05xxxxxxxx",
    "requestText": "وصف المشكلة",
    "reconcileRequestTypeId": 1,
    "attachments": [] // ملفات مرفقة اختيارية
}

النتيجة:
- يتم حفظ الطلب في قاعدة البيانات
- يتم إرسال إشعار لجميع الـ Admins بالبريد الإلكتروني والإشعارات الداخلية

================================================================================
👨‍💼 المرحلة 2: مراجعة الطلب (الإدارة)
================================================================================

الخطوة 2.1: عرض جميع الطلبات
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Admin
الـ Endpoint: GET api/reconcilerequest
الوصف: يقوم الـ Admin بعرض جميع طلبات المصالحة

الخطوة 2.2: عرض تفاصيل طلب محدد
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Admin
الـ Endpoint: GET api/reconcilerequest/{id}
الوصف: يقوم الـ Admin بعرض تفاصيل طلب محدد

الخطوة 2.3: إحالة الطلب للمشرف
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Admin
الحالة الجديدة: AssignedToSupervisor
الـ Endpoint: POST api/reconcilerequest/{id}/assign-supervisor

البيانات المطلوبة:
{
    "supervisorId": 1
}

النتيجة:
- يتم تحديث حالة الطلب
- يتم إرسال إشعار للمشرف المعين

الخطوة 2.4: إلغاء الطلب (اختياري)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Admin
الحالة الجديدة: Cancelled
الـ Endpoint: POST api/reconcilerequest/{id}/cancel
الوصف: يمكن للـ Admin إلغاء الطلب إذا لم يكن مناسباً

================================================================================
👨‍🏫 المرحلة 3: مراجعة المشرف (المشرف)
================================================================================

الخطوة 3.1: عرض الطلبات المحالة للمشرف
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Supervisor أو Admin
الـ Endpoint: GET api/reconcilerequest/supervisor/{supervisorId}
الوصف: يقوم المشرف بعرض الطلبات المحالة إليه

الخطوة 3.2: إحالة الطلب للمستشار
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Supervisor
الحالة الجديدة: AssignedToMediation
الـ Endpoint: POST api/reconcilerequest/{id}/assign-mediation

البيانات المطلوبة:
{
    "mediationId": 1
}

النتيجة:
- يتم تحديث حالة الطلب
- يتم إرسال إشعار للمستشار المعين

الخطوة 3.3: تأكيد مراجعة الطلب
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Supervisor
الحالة الجديدة: SupervisorReviewed
الـ Endpoint: POST api/reconcilerequest/{id}/mark-reviewed
الوصف: يقوم المشرف بتأكيد مراجعته للطلب

النتيجة: يتم إرسال إشعار للـ Admin بأن المشرف قد راجع الطلب

================================================================================
👨‍⚖️ المرحلة 4: تنفيذ المصالحة (المستشار)
================================================================================

الخطوة 4.1: عرض الطلبات المحالة للمستشار
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Mediation أو Admin
الـ Endpoint: GET api/reconcilerequest/mediation/{mediationId}
الوصف: يقوم المستشار بعرض الطلبات المحالة إليه

الخطوة 4.2: بدء تنفيذ الطلب
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Mediation
الحالة الجديدة: InProgress
الـ Endpoint: POST api/reconcilerequest/{id}/start

البيانات المطلوبة:
{
    "consultantNotes": "ملاحظات المستشار"
}

الخطوة 4.3: إكمال تنفيذ المصالحة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Mediation
الحالة الجديدة: PendingSupervisorReview
الـ Endpoint: POST api/reconcilerequest/{id}/complete-execution

البيانات المطلوبة:
{
    "result": "نتيجة المصالحة",
    "consultantNotes": "ملاحظات إضافية"
}

النتيجة: يتم إرسال إشعار للمشرف بأن الطلب بانتظار مراجعته

الخطوة 4.4: إرسال رسائل SMS (اختياري)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Mediation
الـ Endpoint: POST api/reconcilerequest/{id}/send-sms

البيانات المطلوبة:
{
    "message": "نص الرسالة"
}

الخطوة 4.5: عرض نماذج الرسائل
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Mediation
الـ Endpoint: GET api/reconcilerequest/sms-templates
الوصف: يقوم المستشار بعرض نماذج الرسائل الجاهزة

================================================================================
✅ المرحلة 5: إغلاق الطلب (الإدارة)
================================================================================

الخطوة 5.1: إكمال الطلب نهائياً
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
الدور المسؤول: Admin
الحالة النهائية: Completed
الـ Endpoint: POST api/reconcilerequest/{id}/complete
الوصف: يقوم الـ Admin بإغلاق الطلب نهائياً بعد مراجعة العمل

================================================================================
📱 نظام الإشعارات
================================================================================
يتم إرسال إشعارات تلقائية في كل خطوة:
- إشعارات داخلية: للمستخدمين المعنيين
- إشعارات بريد إلكتروني: للـ Admins عند إنشاء طلب جديد

================================================================================
🎯 ملخص تسلسل الحالات
================================================================================
NewRequest → AssignedToSupervisor → AssignedToMediation → InProgress →
PendingSupervisorReview → SupervisorReviewed → Completed

أو في أي مرحلة:
Any Status → Cancelled

================================================================================
📋 ملخص الـ Endpoints حسب الدور
================================================================================

Anonymous:
- POST api/reconcilerequest (إنشاء طلب)

Admin:
- GET api/reconcilerequest (عرض جميع الطلبات)
- GET api/reconcilerequest/{id} (عرض طلب محدد)
- POST api/reconcilerequest/{id}/assign-supervisor (إحالة للمشرف)
- POST api/reconcilerequest/{id}/cancel (إلغاء الطلب)
- POST api/reconcilerequest/{id}/complete (إكمال نهائي)
- DELETE api/reconcilerequest/{id} (حذف الطلب)

Supervisor:
- GET api/reconcilerequest/supervisor/{supervisorId} (طلبات المشرف)
- POST api/reconcilerequest/{id}/assign-mediation (إحالة للمستشار)
- POST api/reconcilerequest/{id}/mark-reviewed (تأكيد المراجعة)

Mediation:
- GET api/reconcilerequest/mediation/{mediationId} (طلبات المستشار)
- POST api/reconcilerequest/{id}/start (بدء التنفيذ)
- POST api/reconcilerequest/{id}/complete-execution (إكمال التنفيذ)
- POST api/reconcilerequest/{id}/send-sms (إرسال SMS)
- GET api/reconcilerequest/sms-templates (نماذج الرسائل)
